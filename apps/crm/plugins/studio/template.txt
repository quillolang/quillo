// map is what the browser runs too (rules only compute); renderers and
// runners run on the server.

let __ID___plugin = {
    id: "__ID__",
    name: "__NAME__",
    description: "What it does, in a sentence.",
    audience: ["merchant", "blogger", "news"],
    blocks: [{ type: "__ID__.callout", label: "Callout", group: "Content", fields: [
        { key: "title", label: "Title", type: "text", required: true },
        { key: "text", label: "Text", type: "textarea" },
        { key: "tone", label: "Tone", type: "select", options: ["info", "success", "warning"] },
    ] }],
    rules: {
        page: [(pg, ctx) => (pg.blocks ?? []) | filter((b) => b.type == "__ID__.callout" && len((b.config ?? {}).title ?? "") > 60)

    },
}

            <b>{c.title}</b>

        </aside>,
}
